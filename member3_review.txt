import logging
from fastapi import APIRouter, UploadFile, File, Form, status
from fastapi.responses import JSONResponse
from pydantic import BaseModel, Field
from typing import Optional

from app.services.voice.transcription import VoiceTranscriptionService, VoiceServiceException, MAX_AUDIO_SIZE_BYTES
from app.services.voice.synthesis import VoiceSynthesisService

logger = logging.getLogger("voice.api")

# Router prefix is /voice so Member 1's main.py can supply prefix="/api"
router = APIRouter(prefix="/voice", tags=["voice"])

transcription_service = VoiceTranscriptionService()
synthesis_service = VoiceSynthesisService()

CHUNK_SIZE = 64 * 1024  # 64 KB chunks for safe streaming read


class VoiceSpeakRequest(BaseModel):
    text: str = Field(..., description="Text to synthesize into speech")
    language: str = Field("en", description="Language code: 'en' or 'te'")


def error_response(code: str, message: str, status_code: int, details: Optional[list] = None) -> JSONResponse:
    return JSONResponse(
        status_code=status_code,
        content={
            "error": {
                "code": code,
                "message": message,
                "details": details or []
            }
        }
    )


@router.post("/transcribe")
async def transcribe_voice(
    file: UploadFile = File(..., description="Audio file to transcribe"),
    language: str = Form("en", description="Target language ('te' or 'en')")
):
    """
    POST /api/voice/transcribe
    Transcribes uploaded audio file to text in Telugu or English.
    """
    try:
        # Enforce 15 MB limit incrementally while reading
        file_bytes = bytearray()
        while True:
            chunk = file.file.read(CHUNK_SIZE) if hasattr(file, "file") else await file.read(CHUNK_SIZE)
            if not chunk:
                break
            file_bytes.extend(chunk)
            if len(file_bytes) > MAX_AUDIO_SIZE_BYTES:
                return error_response(
                    code="PAYLOAD_TOO_LARGE",
                    message=f"Audio file size exceeds the maximum allowed limit of {MAX_AUDIO_SIZE_BYTES // (1024 * 1024)} MB.",
                    status_code=413,
                    details=[{"field": "file", "issue": "Exceeds 15MB limit"}]
                )

        result = await transcription_service.transcribe(
            filename=file.filename,
            content_type=file.content_type,
            file_bytes=bytes(file_bytes),
            language=language
        )
        return result
    except VoiceServiceException as ve:
        return error_response(
            code=ve.code,
            message=ve.message,
            status_code=ve.status_code,
            details=ve.details
        )
    except Exception:
        logger.exception("Unexpected server error in transcribe_voice")
        return error_response(
            code="INTERNAL_SERVER_ERROR",
            message="An internal server error occurred while processing the audio transcription.",
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR
        )


@router.post("/speak")
async def speak_voice(request: VoiceSpeakRequest):
    """
    POST /api/voice/speak
    Synthesizes given text into base64 encoded audio in Telugu or English.
    """
    try:
        result = await synthesis_service.speak(
            text=request.text,
            language=request.language
        )
        return result
    except VoiceServiceException as ve:
        return error_response(
            code=ve.code,
            message=ve.message,
            status_code=ve.status_code,
            details=ve.details
        )
    except Exception:
        logger.exception("Unexpected server error in speak_voice")
        return error_response(
            code="INTERNAL_SERVER_ERROR",
            message="An internal server error occurred while synthesizing speech audio.",
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR
        )

import os

SUPPORTED_LANGUAGES = {"en", "te"}
MAX_AUDIO_SIZE_BYTES = 15 * 1024 * 1024  # 15 MB
ALLOWED_EXTENSIONS = {".wav", ".mp3", ".ogg", ".webm", ".m4a", ".flac", ".aac"}
ALLOWED_MIME_TYPES = {
    "audio/wav", "audio/x-wav", "audio/wave",
    "audio/mpeg", "audio/mp3",
    "audio/ogg", "application/ogg",
    "audio/webm",
    "audio/m4a", "audio/x-m4a", "audio/mp4",
    "audio/flac", "audio/x-flac",
    "audio/aac", "application/octet-stream"
}


class VoiceServiceException(Exception):
    def __init__(self, code: str, message: str, status_code: int = 400, details: Optional[list] = None):
        self.code = code
        self.message = message
        self.status_code = status_code
        self.details = details or []
        super().__init__(self.message)


def is_valid_audio_magic(data: bytes, ext: str) -> bool:
    """
    Verifies actual binary magic bytes / header signatures for supported audio formats.
    Strictly validates audio bytes: arbitrary non-audio bytes are rejected even with a valid extension.
    """
    if len(data) < 4:
        return False

    # WAV / RIFF: starts with 'RIFF', bytes 8..12 are 'WAVE' (or big-endian 'RIFX' ... 'WAVE')
    if len(data) >= 12 and (data.startswith(b"RIFF") or data.startswith(b"RIFX")) and data[8:12] == b"WAVE":
        return True

    # OGG: 'OggS'
    if data.startswith(b"OggS"):
        return True

    # MP3: ID3v2 tag ('ID3') or MPEG frame sync header (0xFF, high 3 bits of next byte set, layer != 0)
    if data.startswith(b"ID3"):
        return True
    if len(data) >= 2 and data[0] == 0xFF and (data[1] & 0xE0) == 0xE0 and (data[1] & 0x06) != 0x00:
        return True

    # FLAC: 'fLaC'
    if data.startswith(b"fLaC"):
        return True

    # MP4 / M4A: ISO base media file with 'ftyp' box at offset 4
    if len(data) >= 8 and data[4:8] == b"ftyp":
        return True

    # WebM / Matroska EBML header: 0x1A 0x45 0xDF 0xA3
    if data.startswith(b"\x1a\x45\xdf\xa3"):
        return True

    # AAC ADTS header: 0xFF followed by 0xF0..0xF1 (syncword 12 bits set)
    if len(data) >= 2 and data[0] == 0xFF and (data[1] & 0xF6) == 0xF0:
        return True

    # Strict: do NOT fall back to checking file extension! Real audio headers are strictly required.
    return False


class VoiceTranscriptionService:
    """Handles audio validation and real STT transcription for Telugu and English."""

    def __init__(self):
        self.sarvam_key = os.getenv("SARVAM_API_KEY")
        self.openai_key = os.getenv("OPENAI_API_KEY")

    def validate_input(self, filename: Optional[str], content_type: Optional[str], file_bytes: bytes, language: str) -> None:
        # Validate language
        if not language or language.lower().strip() not in SUPPORTED_LANGUAGES:
            raise VoiceServiceException(
                code="UNSUPPORTED_LANGUAGE",
                message=f"Unsupported language code '{language}'. Supported languages are: 'te' (Telugu) and 'en' (English).",
                status_code=422,
                details=[{"field": "language", "issue": "Must be 'en' or 'te'"}]
            )

        # Validate file presence and non-empty
        if not file_bytes or len(file_bytes) == 0:
            raise VoiceServiceException(
                code="EMPTY_AUDIO_FILE",
                message="The uploaded audio file is empty. Please provide recorded audio.",
                status_code=400,
                details=[{"field": "file", "issue": "File size is 0 bytes"}]
            )

        # Enforce 15 MB limit
        if len(file_bytes) > MAX_AUDIO_SIZE_BYTES:
            raise VoiceServiceException(
                code="PAYLOAD_TOO_LARGE",
                message=f"Audio file size ({len(file_bytes)} bytes) exceeds the maximum allowed limit of {MAX_AUDIO_SIZE_BYTES // (1024 * 1024)} MB.",
                status_code=413,
                details=[{"field": "file", "issue": "Exceeds 15MB"}]
            )

        # Validate extension or MIME type
        ext = os.path.splitext(filename or "")[1].lower()
        content_type_clean = (content_type or "").split(";")[0].strip().lower()

        is_valid_ext = ext in ALLOWED_EXTENSIONS
        is_valid_mime = content_type_clean in ALLOWED_MIME_TYPES

        if not is_valid_ext and not is_valid_mime:
            raise VoiceServiceException(
                code="INVALID_AUDIO_FORMAT",
                message=f"Unsupported audio format '{ext or content_type_clean}'. Supported formats: wav, mp3, ogg, webm, m4a, flac.",
                status_code=400,
                details=[{"field": "file", "issue": f"Invalid format '{ext or content_type}'"}]
            )

        # Verify minimal audio length
        if len(file_bytes) < 32:
            raise VoiceServiceException(
                code="CORRUPTED_AUDIO_FILE",
                message="Audio file appears corrupted or too brief to contain valid speech data.",
                status_code=400,
                details=[{"field": "file", "issue": "Audio header too small"}]
            )

        # Check binary format header / magic
        if not is_valid_audio_magic(file_bytes, ext):
            raise VoiceServiceException(
                code="INVALID_AUDIO_FORMAT",
                message=f"File content does not match a valid audio format header for '{ext}'.",
                status_code=400,
                details=[{"field": "file", "issue": "Corrupt or invalid audio magic header"}]
            )

    async def transcribe(self, filename: Optional[str], content_type: Optional[str], file_bytes: bytes, language: str = "en") -> Dict[str, Any]:
        """
        Transcribes audio bytes to text in Telugu or English.
        Requires configured real external STT provider (Sarvam AI or OpenAI).
        Never fabricates user speech or confidence scores.
        """
        lang = language.lower().strip()
        self.validate_input(filename, content_type, file_bytes, lang)

        # 1. External Provider: Sarvam AI
        if self.sarvam_key:
            try:
                import httpx
                target_lang_code = "te-IN" if lang == "te" else "en-IN"
                async with httpx.AsyncClient(timeout=30.0) as client:
                    files = {"file": (filename or "audio.wav", file_bytes, content_type or "audio/wav")}
                    data = {"model": "saarika:v1", "language_code": target_lang_code}
                    headers = {"api-subscription-key": self.sarvam_key}
                    response = await client.post("https://api.sarvam.ai/speech-to-text", files=files, data=data, headers=headers)
                    if response.status_code == 200:
                        res_json = response.json()
                        transcript = res_json.get("transcript", "")
                        if transcript:
                            raw_conf = res_json.get("confidence")
                            confidence = None
                            if raw_conf is not None:
                                try:
                                    confidence = round(float(raw_conf), 2)
                                except (ValueError, TypeError):
                                    confidence = None
                            elif "words" in res_json and isinstance(res_json["words"], list):
                                word_confs = [w.get("confidence") for w in res_json["words"] if isinstance(w, dict) and w.get("confidence") is not None]
                                if word_confs:
                                    try:
                                        confidence = round(float(sum(word_confs) / len(word_confs)), 2)
                                    except (ValueError, TypeError, ZeroDivisionError):
                                        confidence = None

                            return {
                                "transcribed_text": transcript,
                                "language": lang,
                                "confidence": confidence
                            }
                    logger.error("Sarvam STT returned status %d: %s", response.status_code, response.text[:200])
            except Exception as e:
                logger.error("Sarvam STT request failed: %s", type(e).__name__)

        # 2. External Provider: OpenAI Whisper
        if self.openai_key:
            try:
                import httpx
                async with httpx.AsyncClient(timeout=30.0) as client:
                    files = {"file": (filename or "audio.wav", file_bytes, content_type or "audio/wav")}
                    data = {"model": "whisper-1", "language": lang}
                    headers = {"Authorization": f"Bearer {self.openai_key}"}
                    response = await client.post("https://api.openai.com/v1/audio/transcriptions", files=files, data=data, headers=headers)
                    if response.status_code == 200:
                        res_json = response.json()
                        transcript = res_json.get("text", "")
                        if transcript:
                            raw_conf = res_json.get("confidence")
                            confidence = None
                            if raw_conf is not None:
                                try:
                                    confidence = round(float(raw_conf), 2)
                                except (ValueError, TypeError):
                                    confidence = None

                            return {
                                "transcribed_text": transcript,
                                "language": lang,
                                "confidence": confidence
                            }
                    logger.error("OpenAI Whisper STT returned status %d: %s", response.status_code, response.text[:200])
            except Exception as e:
                logger.error("OpenAI Whisper request failed: %s", type(e).__name__)

        # 3. No real provider configured or all providers failed
        # Explicitly return 503 Service Unavailable — NEVER invent user speech or fake confidence!
        logger.warning("Voice transcription requested but no STT provider is available or reachable.")
        raise VoiceServiceException(
            code="VOICE_PROVIDER_UNAVAILABLE",
            message="Speech transcription service is currently unavailable. No external STT provider is configured or reachable on the backend.",
            status_code=503,
            details=[{"issue": "Neither SARVAM_API_KEY nor OPENAI_API_KEY is configured or reachable on the server."}]
        )

"""
Voice Synthesis Service for AI Grama Sahayak
Author: Member 3
Contract: POST /api/voice/speak (JSON: text, language)
Returns: { "audio_base64": str, "mime_type": str, "language": str }
"""

import os
import io
import re
import wave
import base64
import logging
from typing import Dict, Any, List, Optional
from .transcription import VoiceServiceException, SUPPORTED_LANGUAGES

logger = logging.getLogger("voice.synthesis")

MAX_TEXT_LENGTH = 4000


def chunk_text(text: str, max_chars: int = 500) -> List[str]:
    """
    Safely splits text into chunks of at most max_chars without dropping any words or characters.
    Splits on sentence boundaries (. ? ! । \n) or whitespace.
    Guarantees 100% of the input text is preserved without silent omission.
    """
    text = text.strip()
    if not text:
        return []
    if len(text) <= max_chars:
        return [text]

    chunks: List[str] = []
    # Split on sentence boundaries while keeping the terminator
    sentences = re.split(r'(?<=[.?!।\n])\s+', text)
    current_chunk = ""

    for s in sentences:
        s = s.strip()
        if not s:
            continue
        if len(s) > max_chars:
            # Long sentence: split on words
            words = s.split()
            for w in words:
                if len(w) > max_chars:
                    # Long unbroken token: slice without losing any characters
                    if current_chunk:
                        chunks.append(current_chunk)
                        current_chunk = ""
                    for i in range(0, len(w), max_chars):
                        sub = w[i:i + max_chars]
                        if len(sub) == max_chars:
                            chunks.append(sub)
                        else:
                            current_chunk = sub
                elif not current_chunk:
                    current_chunk = w
                elif len(current_chunk) + len(w) + 1 <= max_chars:
                    current_chunk += " " + w
                else:
                    chunks.append(current_chunk)
                    current_chunk = w
        else:
            if not current_chunk:
                current_chunk = s
            elif len(current_chunk) + len(s) + 1 <= max_chars:
                current_chunk += " " + s
            else:
                chunks.append(current_chunk)
                current_chunk = s

    if current_chunk:
        chunks.append(current_chunk)

    return chunks


def combine_wav_base64_chunks(b64_chunks: List[str]) -> str:
    """
    Combines multiple base64-encoded WAV audio buffers into a single valid WAV base64 string.
    """
    if not b64_chunks:
        return ""
    if len(b64_chunks) == 1:
        return b64_chunks[0]

    pcm_data = bytearray()
    params = None
    for b64 in b64_chunks:
        try:
            raw = base64.b64decode(b64)
            if len(raw) < 44:
                continue
            with wave.open(io.BytesIO(raw), "rb") as wf:
                if params is None:
                    params = wf.getparams()
                pcm_data.extend(wf.readframes(wf.getnframes()))
        except Exception:
            try:
                raw = base64.b64decode(b64)
                if len(raw) > 44:
                    pcm_data.extend(raw[44:])
            except Exception:
                pass

    if not pcm_data:
        return b64_chunks[0]

    out_buf = io.BytesIO()
    with wave.open(out_buf, "wb") as out_wf:
        if params:
            out_wf.setparams(params)
        else:
            out_wf.setnchannels(1)
            out_wf.setsampwidth(2)
            out_wf.setframerate(8000)
        out_wf.writeframes(bytes(pcm_data))

    return base64.b64encode(out_buf.getvalue()).decode("utf-8")


class VoiceSynthesisService:
    """Handles text validation and real TTS speech synthesis for Telugu and English."""

    def __init__(self):
        self.sarvam_key = os.getenv("SARVAM_API_KEY")
        self.openai_key = os.getenv("OPENAI_API_KEY")

    def validate_input(self, text: str, language: str) -> None:
        if not language or language.lower().strip() not in SUPPORTED_LANGUAGES:
            raise VoiceServiceException(
                code="UNSUPPORTED_LANGUAGE",
                message=f"Unsupported language code '{language}'. Supported languages are: 'te' (Telugu) and 'en' (English).",
                status_code=422,
                details=[{"field": "language", "issue": "Must be 'en' or 'te'"}]
            )

        if not text or not text.strip():
            raise VoiceServiceException(
                code="EMPTY_TEXT",
                message="Text to synthesize cannot be empty. Please provide text for speech generation.",
                status_code=400,
                details=[{"field": "text", "issue": "Text must not be empty or whitespace"}]
            )

        if len(text) > MAX_TEXT_LENGTH:
            raise VoiceServiceException(
                code="TEXT_TOO_LONG",
                message=f"Text length ({len(text)} characters) exceeds maximum limit of {MAX_TEXT_LENGTH} characters.",
                status_code=400,
                details=[{"field": "text", "issue": f"Exceeds {MAX_TEXT_LENGTH} characters"}]
            )

    async def speak(self, text: str, language: str = "en") -> Dict[str, Any]:
        """
        Synthesizes text to speech in Telugu or English.
        Requires configured real external TTS provider (Sarvam AI Bulbul or OpenAI TTS).
        Never returns fake audio chimes or synthetic filler as spoken output.
        Never silently truncates or omits text.
        """
        lang = language.lower().strip()
        self.validate_input(text, lang)

        # 1. External Provider: Sarvam AI (Native support for Telugu & Indian English)
        if self.sarvam_key:
            try:
                import httpx
                target_lang_code = "te-IN" if lang == "te" else "en-IN"
                chunks = chunk_text(text, max_chars=500)
                if not chunks:
                    chunks = [text]

                async with httpx.AsyncClient(timeout=45.0) as client:
                    payload = {
                        "inputs": chunks,
                        "target_language_code": target_lang_code,
                        "speaker": "meera",
                        "pitch": 0,
                        "pace": 1.0,
                        "speech_sample_rate": 8000,
                        "enable_preprocessing": True,
                        "model": "bulbul:v1"
                    }
                    headers = {
                        "api-subscription-key": self.sarvam_key,
                        "Content-Type": "application/json"
                    }
                    response = await client.post("https://api.sarvam.ai/text-to-speech", json=payload, headers=headers)
                    if response.status_code == 200:
                        res_json = response.json()
                        audios = res_json.get("audios", [])
                        if audios and len(audios) > 0:
                            combined_audio = combine_wav_base64_chunks(audios)
                            return {
                                "audio_base64": combined_audio,
                                "mime_type": "audio/wav",
                                "language": lang
                            }
                    logger.error("Sarvam TTS returned status %d: %s", response.status_code, response.text[:200])
            except Exception as e:
                logger.error("Sarvam TTS request failed: %s", type(e).__name__)

        # 2. External Provider: OpenAI TTS
        if self.openai_key:
            try:
                import httpx
                async with httpx.AsyncClient(timeout=45.0) as client:
                    payload = {
                        "model": "tts-1",
                        "input": text,
                        "voice": "nova"
                    }
                    headers = {
                        "Authorization": f"Bearer {self.openai_key}",
                        "Content-Type": "application/json"
                    }
                    response = await client.post("https://api.openai.com/v1/audio/speech", json=payload, headers=headers)
                    if response.status_code == 200:
                        audio_b64 = base64.b64encode(response.content).decode("utf-8")
                        return {
                            "audio_base64": audio_b64,
                            "mime_type": "audio/mpeg",
                            "language": lang
                        }
                    logger.error("OpenAI TTS returned status %d: %s", response.status_code, response.text[:200])
            except Exception as e:
                logger.error("OpenAI TTS request failed: %s", type(e).__name__)

        # 3. No real provider configured or all providers failed
        # Explicitly return 503 Service Unavailable — NEVER return chimes or fake audio!
        logger.warning("Voice synthesis requested but no TTS provider is available or reachable.")
        raise VoiceServiceException(
            code="VOICE_PROVIDER_UNAVAILABLE",
            message="Text-to-speech service is currently unavailable. No external TTS provider is configured or reachable on the backend.",
            status_code=503,
            details=[{"issue": "Neither SARVAM_API_KEY nor OPENAI_API_KEY is configured or reachable on the server."}]
        )

export class ApiException extends Error {
  code: string;
  status: number;
  details?: unknown[];

  constructor(message: string, code: string = "API_ERROR", status: number = 500, details?: unknown[]) {
    super(message);
    this.name = "ApiException";
    this.code = code;
    this.status = status;
    this.details = details;

async function handleResponse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    let errorCode = "HTTP_ERROR";
    let errorMessage = `Server returned error status ${res.status} (${res.statusText})`;
    let details: unknown[] | undefined;

    try {
      const errorJson = await res.json();
      if (errorJson?.error) {
        errorCode = errorJson.error.code || errorCode;
        errorMessage = errorJson.error.message || errorMessage;
        details = errorJson.error.details;
      }
    } catch {
      // response body was not valid JSON
    }

    throw new ApiException(errorMessage, errorCode, res.status, details);
  }

  return (await res.json()) as T;
}

export async function transcribeAudio(
  audioBlob: Blob,
  language: "en" | "te" = "en",
  signal?: AbortSignal
): Promise<VoiceTranscribeResponse> {

  const formData = new FormData();

/**
 * Fetch nearby government services (Member 4)
 */
export async function getNearbyServices(
  params: {
    service_type?: string;
    district?: string;
    latitude?: number;
    longitude?: number;
  },
  signal?: AbortSignal
): Promise<NearbyServicesResponse> {
  const query = new URLSearchParams();
  if (params.service_type) query.append("service_type", params.service_type);
  if (params.district) query.append("district", params.district);
  if (params.latitude !== undefined && params.longitude !== undefined) {
    query.append("latitude", params.latitude.toString());
    query.append("longitude", params.longitude.toString());
  }

  const url = `${API_BASE_URL}/api/services/nearby?${query.toString()}`;
  const res = await fetch(url, {
    method: "GET",
    signal,
  });

  return handleResponse<NearbyServicesResponse>(res);
}

type SpeechRecognitionAlternativeLike = {
  transcript: string;
};

type SpeechRecognitionResultLike = {
  0: SpeechRecognitionAlternativeLike;
};

type SpeechRecognitionEventLike = {
  results: ArrayLike<SpeechRecognitionResultLike>;
};

type SpeechRecognitionErrorLike = {
  error: string;
};

type SpeechRecognitionLike = {
  lang: string;
  continuous: boolean;
  interimResults: boolean;
  onstart: (() => void) | null;
  onresult: ((event: SpeechRecognitionEventLike) => void) | null;
  onerror: ((event: SpeechRecognitionErrorLike) => void) | null;
  onend: (() => void) | null;
  start: () => void;
  stop: () => void;
};

type SpeechRecognitionConstructor = new () => SpeechRecognitionLike;

type WindowWithSpeechRecognition = Window & {
  SpeechRecognition?: SpeechRecognitionConstructor;
  webkitSpeechRecognition?: SpeechRecognitionConstructor;
};

type ChatMessage = {
  id: number;
  sender: "user" | "assistant";
  text: string;
  sources?: Source[];
  safetyNotice?: string | null;
  followUps?: string[];

type ActiveTab = "chat" | "documents" | "services" | "grievances";

export default function Home() {
  const [activeTab, setActiveTab] = useState<ActiveTab>("chat");
  const [language, setLanguage] = useState("Telugu");

  const [voiceMessage, setVoiceMessage] = useState("");
  const [isListening, setIsListening] = useState(false);

  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [isThinking, setIsThinking] = useState(false);
  const [backendNotice, setBackendNotice] = useState<string | null>(null);

  const recognitionRef = useRef<SpeechRecognitionLike | null>(null);
  const chatEndRef = useRef<HTMLDivElement>(null);
  const abortControllerRef = useRef<AbortController | null>(null);
  const conversationIdRef = useRef<string | null>(null);
  const lastFailedMessageRef = useRef<string | null>(null);
  const messageIdCounterRef = useRef(1);

  // Automatically scroll to the latest chat content
  useEffect(() => {
    if (activeTab === "chat") {
      chatEndRef.current?.scrollIntoView({
        behavior: "smooth",
        block: "end",
      });
    }
  }, [messages, isThinking, activeTab]);

  // Clean up any in-flight requests when the component unmounts
  useEffect(() => {
    return () => {
      if (abortControllerRef.current) {
        abortControllerRef.current.abort();
      }
    };
  }, []);

  /**
   * Browser Speech Recognition (Web Speech API).
   * Note: Browser SpeechRecognition may use browser/vendor network services
   * (e.g. Google speech servers in Chrome) and is not guaranteed to operate offline.
   */
  const startVoiceInput = () => {
    const browserWindow = window as WindowWithSpeechRecognition;

    const SpeechRecognitionAPI =
      browserWindow.SpeechRecognition ||
      browserWindow.webkitSpeechRecognition;

    if (!SpeechRecognitionAPI) {
      setVoiceMessage(
        "Speech recognition is not supported in this browser. Please try Google Chrome."
      );
      return;
    }

    if (isListening) {
      recognitionRef.current?.stop();
      setIsListening(false);
      return;
    }

    const recognition = new SpeechRecognitionAPI();

    recognition.lang = language === "Telugu" ? "te-IN" : "en-IN";

    recognition.continuous = false;
    recognition.interimResults = false;

    recognition.onstart = () => {
      setIsListening(true);
      setVoiceMessage("Listening... Speak your question now 🎤");
    };

    recognition.onresult = (event) => {
      const transcript = event.results[0][0].transcript;

      setQuestion(transcript);
      setVoiceMessage("Your speech was converted to text successfully! ✓");
    };

    recognition.onerror = (event) => {
      setIsListening(false);

      if (event.error === "not-allowed") {
        setVoiceMessage(
          "Microphone permission denied. Please allow microphone access in your browser."
        );
      } else if (event.error === "no-speech") {
        setVoiceMessage(
          "No speech detected. Please tap the microphone and try again."
        );
      } else if (event.error === "network") {
        setVoiceMessage(
          "Network error occurred during speech recognition. Please try again."
        );
      } else {
        setVoiceMessage(
          "Something went wrong with voice recognition. Please try again."
        );
      }
    };

    recognition.onend = () => {
      setIsListening(false);
    };

    recognitionRef.current = recognition;

    try {
      recognition.start();

      setIsListening(false);
      setVoiceMessage("Unable to start microphone. Please try again.");

      text: trimmedQuestion,

        id: messageIdCounterRef.current++,
        sender: "assistant",
        text:
          language === "Telugu"
            ? "క్షమించండి, గ్రామ సహాయక్ చాట్ సర్వీస్ ప్రస్తుతం అందుబాటులో లేదు (/api/chat). దయచేసి కాసేపటి తర్వాత 'మళ్లీ ప్రయత్నించండి' బటన్ నొక్కండి."

        ...previousMessages,
        unavailableMessage,

"use client";

import { useState, useRef, useEffect } from "react";
import { synthesizeSpeech } from "../services/api";

interface AudioSpeakButtonProps {
  text: string;
  language: string; // "Telugu" | "English"
}

export default function AudioSpeakButton({ text, language }: AudioSpeakButtonProps) {
  const [isPlaying, setIsPlaying] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const [statusMessage, setStatusMessage] = useState<string | null>(null);
  const audioRef = useRef<HTMLAudioElement | null>(null);

  const isTe = language === "Telugu";
  const langCode: "te" | "en" = isTe ? "te" : "en";

  const stopAudio = () => {
    if (audioRef.current) {
      audioRef.current.pause();
      audioRef.current.currentTime = 0;
      audioRef.current = null;
    }
    if (typeof window !== "undefined" && window.speechSynthesis) {
      window.speechSynthesis.cancel();
    }
    setIsPlaying(false);
    setIsLoading(false);
  };

  useEffect(() => {
    return () => {
      stopAudio();
    };
  }, []);

  const handleToggleSpeak = async () => {
    if (isPlaying || isLoading) {
      stopAudio();
      return;
    }

    setIsLoading(true);
    setStatusMessage(null);

    try {
      // 1. Try backend voice synthesis endpoint
      const res = await synthesizeSpeech(text, langCode);
      if (res?.audio_base64) {
        const audioSrc = `data:${res.mime_type || "audio/wav"};base64,${res.audio_base64}`;
        const audio = new Audio(audioSrc);
        audioRef.current = audio;

        audio.onplay = () => {
          setIsLoading(false);
          setIsPlaying(true);
        };

        audio.onended = () => {
          setIsPlaying(false);
          audioRef.current = null;
        };

        audio.onerror = () => {
          fallbackBrowserSpeech();
        };

        await audio.play();
        return;
      } else {
        fallbackBrowserSpeech();
      }
    } catch {
      // Backend TTS unavailable (e.g. 503 or offline) — fallback to browser speech synthesis
      fallbackBrowserSpeech();
    }
  };

  const fallbackBrowserSpeech = () => {
    if (typeof window === "undefined" || !("speechSynthesis" in window)) {
      setIsLoading(false);
      setIsPlaying(false);
      setStatusMessage(
        isTe
          ? "ఈ బ్రౌజర్‌లో వాయిస్ ప్లేబ్యాక్ అందుబాటులో లేదు."
          : "Voice synthesis is not supported in this browser."
      );
      return;
    }

    window.speechSynthesis.cancel();

    // Check available voices
    const voices = window.speechSynthesis.getVoices();
    const targetLang = isTe ? "te-IN" : "en-IN";
    const voice = voices.find((v) => v.lang.startsWith(isTe ? "te" : "en"));

    if (isTe && voices.length > 0 && !voice) {
      // Browser has voices but no Telugu voice pack installed
      setIsLoading(false);
      setIsPlaying(false);
      setStatusMessage(
        "ఈ బ్రౌజర్ లేదా పరికరంలో తెలుగు వాయిస్ ప్యాక్ అందుబాటులో లేదు."
      );
      return;
    }

    const utterance = new SpeechSynthesisUtterance(text);
    utterance.lang = targetLang;
    if (voice) utterance.voice = voice;
    utterance.rate = 0.95;

    utterance.onstart = () => {
      setIsLoading(false);
      setIsPlaying(true);
    };

    utterance.onend = () => {
      setIsPlaying(false);
    };

    utterance.onerror = () => {
      setIsLoading(false);
      setIsPlaying(false);
      setStatusMessage(
        isTe
          ? "ఆడియో ప్లేబ్యాక్ విఫలమైంది."
          : "Audio playback encountered an error."
      );
    };

    window.speechSynthesis.speak(utterance);
  };

  return (
    <div className="audio-speak-container">
      <button
        className={`audio-speak-btn ${isPlaying ? "playing" : ""}`}
        onClick={handleToggleSpeak}
        disabled={isLoading}
        aria-label={isPlaying ? "Stop speech" : "Listen to response"}
        title={isPlaying ? "Stop audio" : "Listen to audio response"}
      >
        {isLoading ? (
          <span>⏳ {isTe ? "సిద్ధమవుతోంది..." : "Loading..."}</span>
        ) : isPlaying ? (
          <span>⏹️ {isTe ? "ఆపండి" : "Stop"}</span>
        ) : (
          <span>🔊 {isTe ? "వినండి" : "Listen"}</span>
        )}
      </button>
      {statusMessage && (
        <span className="audio-status-msg" role="status">
          {statusMessage}
        </span>
      )}
    </div>
  );
}

import { useState } from "react";
import { draftGrievance, GrievanceDraftResponse, ApiException } from "../services/api";

interface GrievanceDraftingProps {

  { id: "electricity", labelEn: "Electricity & Power Outage", labelTe: "విద్యుత్ సరఫరా & స్తంభాల సమస్య" },
  { id: "roads", labelEn: "Roads & Street Infrastructure", labelTe: "రహదారులు & వీధి లైట్ల మరమ్మత్తు" },
  { id: "ration", labelEn: "Ration & PDS Distribution", labelTe: "రేషన్ & పౌరసరఫరాల సమస్య" },
  { id: "drainage", labelEn: "Sanitation & Drainage Cleaning", labelTe: "పారిశుధ్యం & డ్రైనేజీ సమస్య" },
  { id: "welfare", labelEn: "Welfare Scheme Delay / Issue", labelTe: "సంక్షేమ పథకం మంజూరులో ఆలస్యం" },
  { id: "other", labelEn: "Other Citizen Grievance", labelTe: "ఇతర ప్రజా ఫిర్యాదు" },
];

export default function GrievanceDrafting({ language }: GrievanceDraftingProps) {
  const isTe = language === "Telugu";

  const [category, setCategory] = useState<string>("water_supply");
  const [description, setDescription] = useState<string>("");
  const [district, setDistrict] = useState<string>("");
  const [mandal, setMandal] = useState<string>("");
  const [draftLang, setDraftLang] = useState<"te" | "en">(isTe ? "te" : "en");
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [error, setError] = useState<string | null>(null);
  const [draftResult, setDraftResult] = useState<GrievanceDraftResponse | null>(null);
  const [editableBody, setEditableBody] = useState<string>("");
  const [copied, setCopied] = useState<boolean>(false);

  const handleGenerateDraft = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!description.trim()) {
      setError(
        isTe
          ? "దయచేసి మీ సమస్య వివరాలను వివరించండి."
          : "Please describe your grievance in a few sentences."
      );
      return;
    }

    if (!district.trim()) {
      setError(
        isTe
          ? "దయచేసి మీ జిల్లా పేరును నమోదు చేయండి."
          : "Please enter your district name."
      );
      return;
    }

    if (!mandal.trim()) {
      setError(
        isTe
          ? "దయచేసి మీ మండలం లేదా గ్రామం పేరును నమోదు చేయండి."
          : "Please enter your mandal or village name."
      );
      return;
    }

    setIsLoading(true);
    setError(null);
    setCopied(false);

    try {
      const response = await draftGrievance({
        category,
        description: description.trim(),
        district: district.trim(),
        mandal: mandal.trim(),
        language: draftLang,
      });

      setDraftResult(response);
      setEditableBody(response.complaint_body);
    } catch (err: unknown) {
      if (err instanceof ApiException) {
        if (err.status === 404 || err.status === 503 || err.code === "HTTP_ERROR") {
          setError(
            isTe
              ? "ఫిర్యాదు ముసాయిదా సర్వరు ప్రస్తుతం ఆఫ్‌లైన్‌లో ఉంది. (Member 4 బాధ్యత: /api/grievances/draft)."
              : `Grievance drafting backend service is currently offline or unreachable at /api/grievances/draft. (${err.message})`
          );
        } else {
          setError(err.message);
        }
      } else {
        setError(
          isTe
            ? "సర్వరుతో అనుసంధానం కాలేకపోయింది. దయచేసి బ్యాకెండ్ సర్వరు నడుస్తుందో లేదో తనిఖీ చేయండి."
            : "Could not connect to backend server. Please verify the FastAPI backend is running."
        );
      }
    } finally {
      setIsLoading(false);
    }
  };

  const handleCopy = () => {
    if (!draftResult) return;
    const fullText = `${isTe ? "విషయం:" : "Subject:"} ${draftResult.subject}\n\n${editableBody}`;
    navigator.clipboard.writeText(fullText).then(() => {
      setCopied(true);
      setTimeout(() => setCopied(false), 3000);
    });
  };

  return (
    <div className="tool-card">
      <div className="tool-header">
        <div className="tool-icon">📝</div>
        <div>
          <h3>{isTe ? "ప్రజా ఫిర్యాదు ముసాయిదా సహాయకుడు" : "Citizen Grievance Drafting"}</h3>
          <p className="tool-subtitle">
            {isTe
              ? "సమస్యను వివరించండి — అధికారిక దరఖాస్తు ముసాయిదాను సిద్ధం చేసుకోండి"
              : "Describe your village issue and generate a formal petition for submission"}
          </p>
        </div>
      </div>

      <form onSubmit={handleGenerateDraft} className="grievance-form">
        <div className="form-grid">
          <div className="form-group">
            <label className="field-label" htmlFor="grievance-category">
              {isTe ? "సమస్య విభాగం" : "Category"}
            </label>
            <select
              id="grievance-category"
              className="form-select"
              value={category}
              onChange={(e) => setCategory(e.target.value)}
            >
              {GRIEVANCE_CATEGORIES.map((c) => (
                <option key={c.id} value={c.id}>
                  {isTe ? c.labelTe : c.labelEn}
                </option>
              ))}
            </select>
          </div>

          <div className="form-group">
            <label className="field-label" htmlFor="grievance-lang">
              {isTe ? "ముసాయిదా భాష" : "Draft Language"}
            </label>
            <select
              id="grievance-lang"
              className="form-select"
              value={draftLang}
              onChange={(e) => setDraftLang(e.target.value as "te" | "en")}
            >
              <option value="te">తెలుగు (Telugu)</option>
              <option value="en">English</option>
            </select>
          </div>

          <div className="form-group">
            <label className="field-label" htmlFor="grievance-district">
              {isTe ? "జిల్లా *" : "District *"}
            </label>
            <input
              id="grievance-district"
              type="text"
              className="form-input"
              value={district}
              onChange={(e) => setDistrict(e.target.value)}
              placeholder={isTe ? "ఉదా: కాకినాడ (తప్పనిసరి)" : "e.g. Kakinada (required)"}
              required
            />
          </div>

          <div className="form-group">
            <label className="field-label" htmlFor="grievance-mandal">
              {isTe ? "మండలం / గ్రామం *" : "Mandal / Village *"}
            </label>
            <input
              id="grievance-mandal"
              type="text"
              className="form-input"
              value={mandal}
              onChange={(e) => setMandal(e.target.value)}
              placeholder={isTe ? "ఉదా: సామర్లకోట మండలం (తప్పనిసరి)" : "e.g. Samalkota Mandal (required)"}
              required
            />
          </div>

          <label className="field-label" htmlFor="grievance-desc">

            rows={4}

            onChange={(e) => setDescription(e.target.value)}
            placeholder={
              isTe

  );
